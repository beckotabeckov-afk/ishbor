// ═══════════════════════════════════════════════════════════════════
//  TTS — Web Speech API (browser built-in, free, no API)
// ═══════════════════════════════════════════════════════════════════

function speak(text) {
  return new Promise((resolve) => {
    window.speechSynthesis.cancel();
    const utt = new SpeechSynthesisUtterance(text);
    utt.lang = 'uz-UZ';
    utt.rate = 0.95;
    utt.pitch = 1.05;
    // Try to find a good voice
    const voices = window.speechSynthesis.getVoices();
    const uz = voices.find(v => v.lang.startsWith('uz'));
    const ru = voices.find(v => v.lang.startsWith('ru'));
    const en = voices.find(v => v.lang.startsWith('en') && v.name.includes('Female'));
    utt.voice = uz || ru || en || voices[0] || null;
    utt.onend = resolve;
    utt.onerror = resolve;
    window.speechSynthesis.speak(utt);
  });
}

// ═══════════════════════════════════════════════════════════════════
//  STT — Web Speech API (browser built-in)
// ═══════════════════════════════════════════════════════════════════

function startRecognition(onResult, onEnd) {
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SR) { alert("Bu brauzer ovoz tanishni qo'llab-quvvatlamaydi. Chrome ishlating."); return; }
  recognition = new SR();
  recognition.lang = 'uz-UZ';
  recognition.continuous = false;
  recognition.interimResults = true;
  recognition.maxAlternatives = 1;

  recognition.onresult = (e) => {
    let interim = '', final = '';
    for (let r of e.results) {
      if (r.isFinal) final += r[0].transcript;
      else interim += r[0].transcript;
    }
    setSubtitle(`"${final || interim}"`);
    if (final) onResult(final);
  };
  recognition.onend = onEnd;
  recognition.onerror = (e) => { console.warn('STT error:', e.error); onEnd(); };
  recognition.start();
}

  if (!ANTHROPIC_KEY) {

Current step: ${step}
Current CV data: ${JSON.stringify(cv)}
${vacContext}

User said: "${userText}"

Extract information for step "${step}" and return JSON only. No explanations.

Rules:

Return ONLY valid JSON:

Step-specific extraction:
- GREETING: any yes/okay/ha → CONTINUE

- REGION: match to Uzbekistan region → region field
- DISTRICT: extract district name → district field
- EDUCATION: extract level → education field

- JOB_TYPE: extract job type → job_type field

- SALARY: extract salary as string → expected_salary field
- PHONE: extract phone number → phone field

- CONFIRMATION: yes/ha/да → CONTINUE, no → BACK`;

  try {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': ANTHROPIC_KEY,
        'anthropic-version': '2023-06-01',
        'anthropic-dangerous-direct-browser-access': 'true'
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-6',

        messages: [{ role: 'user', content: prompt }]
      })
    });

    const data = await res.json();

function matchVacancies() {
  const scored = VACANCIES.map(v => {
    let score = 0;
    cv.skills.forEach(skill => {
      const s = skill.toLowerCase();
      if (v.skills.some(vs => vs.toLowerCase().includes(s) || s.includes(vs.toLowerCase()))) score += 3;
    });
    if (cv.preferred_region && v.region.toLowerCase().includes(cv.preferred_region.toLowerCase())) score += 2;
    const salNum = parseInt((cv.expected_salary || '').replace(/\D/g,'')) || 0;
    const vacNum = parseInt(v.salary.replace(/\D/g,'')) || 0;
    if (salNum > 0 && vacNum >= salNum * 0.8) score += 1;
    return { ...v, score };
  });
  return scored.sort((a,b) => b.score - a.score).slice(0, 3);
}

  navigator.mediaDevices.getUserMedia({ audio: true, video: false })
    .then(function(stream) {
      // Release immediately — we'll re-request when actually recording
      stream.getTracks().forEach(function(t) { t.stop(); });

      var SR = window.SpeechRecognition || window.webkitSpeechRecognition;

      mainApp.classList.remove('hidden');

async function runStep(idx) {
  stepIndex = idx;
  updateProgress();

  const step = STEPS[idx];
  const q = QUESTIONS[step]
    .replace('[Name]', cv.name || "Do'stim")
    .replace('[Job title]', cv.selected_vacancy?.title || 'ish')
    .replace('[Company]', cv.selected_vacancy?.company || 'kompaniya');

  // PHOTO step
  if (step === 'PHOTO') {
    setEmotion('HAPPY');
    setStatus('speaking', 'Gapiryapman...');
    setSubtitle(q);
    await speak(q);
    await startCamera();
    await runCountdown();
    return;
  }

  // FAREWELL step
  if (step === 'FAREWELL') {
    setEmotion('CELEBRATING');
    setStatus('speaking', 'Yakunlash...');
    setSubtitle(q);
    document.getElementById('micArea').style.display = 'none';
    launchConfetti();
    renderCVSummary();
    await speak(q);
    return;
  }

  // MATCHING step
  if (step === 'MATCHING') {
    setEmotion('THINKING');
    setStatus('thinking', 'Qidiryapman...');
    setSubtitle('Sizga mos vakansiyalar qidirilmoqda...');
    await speak(QUESTIONS.MATCHING);
    matches = matchVacancies();
    renderVacancies();
    await runStep(STEPS.indexOf('VACANCY_SELECTION'));
    return;
  }

  // Normal steps: speak question then listen
  setEmotion('HAPPY');
  setStatus('speaking', 'Gapiryapman...');
  setSubtitle(q);
  showWaveform(false);
  await speak(q);
  await listenAndProcess();
}

async function listenAndProcess() {
  return new Promise(async (resolve) => {
    setEmotion('LISTENING');
    setStatus('listening', 'Tinglayapman...');
    setSubtitle('Gapiring...');
    showWaveform(true);
    document.getElementById('micBtn').className = 'mic-btn recording';
    document.getElementById('micBtn').textContent = '⏹';
    document.getElementById('micHint').textContent = 'Tugmani bosib to\'xtating';
    isListening = true;

    let finalText = '';
    let recognized = false;

    startInactivityTimer();

    startRecognition(
      (text) => { finalText = text; recognized = true; },
      async () => {
        clearInactivityTimer();
        isListening = false;
        document.getElementById('micBtn').className = 'mic-btn idle';
        document.getElementById('micBtn').textContent = '🎙️';
        document.getElementById('micHint').textContent = 'Bosib gapiring';
        showWaveform(false);

        if (!finalText.trim()) {
          setEmotion('CONFUSED');
          setStatus('idle', 'Eshitmadim');
          await speak("Kechirasiz, eshitmadim. Yana bir marta ayta olasizmi?");
          resolve();
          await listenAndProcess();
          return;
        }

        await processUserInput(finalText);
        resolve();
      }
    );
  });
}

function toggleMic() {
  if (isListening && recognition) {
    recognition.stop();
  } else if (!isSpeaking && !isProcessing) {
    listenAndProcess();
  }
}

async function processUserInput(text) {
  setEmotion('THINKING');
  setStatus('thinking', 'O\'ylayapman...');
  setSubtitle('Tahlil qilyapman...');
  isProcessing = true;

  const result = await callClaude(text);

  isProcessing = false;

  if (!result) {
    setEmotion('CONFUSED');
    await speak("Xatolik yuz berdi. Qaytadan urinib ko'ramiz.");
    await listenAndProcess();
    return;
  }

  if (result.nextAction === 'BACK') { goBack(); return; }
  if (result.nextAction === 'STOP') {
    setEmotion('CONFUSED');
    await speak("Davom etamizmi?");
    await listenAndProcess();
    return;
  }

  // Merge CV (don't overwrite non-empty fields with empty)
  const u = result.updatedCV || {};
  for (const [k, v] of Object.entries(u)) {
    if (v === null || v === undefined) continue;
    if (typeof v === 'string' && !v.trim()) continue;
    if (Array.isArray(v) && v.length === 0) continue;
    cv[k] = v;
  }

  // Vacancy selection
  if (result.selected_vacancy_index != null && matches[result.selected_vacancy_index]) {
    cv.selected_vacancy = matches[result.selected_vacancy_index];
    renderVacancies();
  }

  // Speak confirmation
  if (result.confirmationText) {
    setEmotion('HAPPY');
    setStatus('speaking', 'Tasdiq...');
    setSubtitle(result.confirmationText);
    await speak(result.confirmationText);
  }

  // Move to next step
  const next = stepIndex + 1;
  if (next < STEPS.length) {
    await runStep(next);
  }
}

function goBack() {
  if (stepIndex > 0) runStep(stepIndex - 1);
}

function startInactivityTimer() {
  clearInactivityTimer();
  inactivityTimer = setTimeout(async () => {
    if (isListening) {
      setEmotion('CONFUSED');
      if (recognition) recognition.stop();
      await speak("Siz hali o'ylamoqdasizmi? Shoshilmang, qayta bosing.");
    }
  }, 15000);
}

function clearInactivityTimer() {
  if (inactivityTimer) { clearTimeout(inactivityTimer); inactivityTimer = null; }
}

async function startCamera() {
  const wrap = document.getElementById('cameraWrap');
  wrap.classList.remove('hidden');
  document.getElementById('micArea').style.display = 'none';
  try {
    camStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode:'user' } });
    document.getElementById('camera-video').srcObject = camStream;
  } catch(e) {
    wrap.classList.add('hidden');
  }
}

function runCountdown() {
  return new Promise((resolve) => {
    const overlay = document.getElementById('countdownOverlay');
    const numEl = document.getElementById('countdownNum');
    overlay.classList.remove('hidden');
    let n = 3;
    numEl.textContent = n;
    const iv = setInterval(() => {
      n--;
      if (n > 0) {
        numEl.textContent = n;
        numEl.style.animation = 'none';
        setTimeout(() => numEl.style.animation = 'countIn 0.5s cubic-bezier(0.34,1.56,0.64,1)', 10);
      } else {
        clearInterval(iv);
        capturePhoto();
        resolve();
      }
    }, 1000);
  });
}

async function capturePhoto() {
  const video = document.getElementById('camera-video');
  const canvas = document.createElement('canvas');
  canvas.width = video.videoWidth || 640;
  canvas.height = video.videoHeight || 480;
  canvas.getContext('2d').drawImage(video, 0, 0);
  photoDataUrl = canvas.toDataURL('image/jpeg', 0.9);
  cv.photo_taken = true;

  if (camStream) camStream.getTracks().forEach(t => t.stop());

  document.getElementById('cameraWrap').classList.add('hidden');
  document.getElementById('countdownOverlay').classList.add('hidden');
  const photoEl = document.getElementById('captured-photo');
  photoEl.src = photoDataUrl;
  photoEl.style.display = 'block';

  setEmotion('CELEBRATING');
  launchConfetti();
  await speak("Rahmat! Siz juda chiroyli jilmaydingiz!");
  await runStep(STEPS.indexOf('FAREWELL'));
}

function launchConfetti() {
  const canvas = document.getElementById('confetti-canvas');
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
  const ctx = canvas.getContext('2d');
  const pieces = Array.from({length:120}, () => ({
    x: Math.random() * canvas.width,
    y: -20,
    w: 8 + Math.random() * 8,
    h: 6 + Math.random() * 6,
    color: ['#FFD93D','#6BCB77','#4D96FF','#FF6B9D','#FF9F1C'][Math.floor(Math.random()*5)],
    speed: 2 + Math.random() * 4,
    angle: Math.random() * Math.PI * 2,
    spin: (Math.random() - 0.5) * 0.2,
    drift: (Math.random() - 0.5) * 2
  }));

  let frame;
  function draw() {
    ctx.clearRect(0,0,canvas.width,canvas.height);
    let active = 0;
    pieces.forEach(p => {
      if (p.y > canvas.height + 20) return;
      active++;
      p.y += p.speed;
      p.x += p.drift;
      p.angle += p.spin;
      ctx.save();
      ctx.translate(p.x, p.y);
      ctx.rotate(p.angle);
      ctx.fillStyle = p.color;
      ctx.fillRect(-p.w/2, -p.h/2, p.w, p.h);
      ctx.restore();
    });
    if (active > 0) frame = requestAnimationFrame(draw);
    else ctx.clearRect(0,0,canvas.width,canvas.height);
  }
  draw();
}
